REPLACE PROCEDURE GPROD1P_PP.TBL_DELTA	(   IN I_RUN_ID DECIMAL(38),
											IN I_SCRIPT_ID INTEGER,
											IN I_SOURCE_NAME VARCHAR(50),
											IN I_SCHEMA VARCHAR(100),
											IN I_TBL_NAME VARCHAR(100),
											IN I_SQL_STMT CLOB CHARACTER SET UNICODE,
											IN I_PRIM_INDEX CLOB CHARACTER SET UNICODE,
											IN I_SECD_INDEX CLOB CHARACTER SET UNICODE,
											IN I_NOT_NULLABLE_COL CLOB CHARACTER SET UNICODE,
											IN I_DELTA   INTEGER ,
											IN I_COLS_EXCLUDE_FROM_DELTA VARCHAR(500),
											IN I_TARGET_TABLE_KEYS  VARCHAR(500),
											OUT O_RETURN_CODE INTEGER,
											OUT O_RETURN_MSG  VARCHAR(10000),
											OUT O_ROWS_COUNT INTEGER
									 )
/*
###################################################################################################################################################################
------------------------------------------------------------------------------------------------------------------------------------------------------------------
#  Information
------------------------------------------------------------------------------------------------------------------------------------------------------------------
#  Filename  : TBL_DELTA.SQL
#  File Type  : SQL file
------------------------------------------------------------------------------------------------------------------------------------------------------------------
#  Purpose  : This Stored Procedure is used to search 
------------------------------------------------------------------------------------------------------------------------------------------------------------------
#  Version History
------------------------------------------------------------------------------------------------------------------------------------------------------------------
#	Version 		| Date			          | Author         			              | COMMENST
#	01.0			| 12  APR, 2023  	| AHMED EHAB					| INITIAL.

------------------------------------------------------------------------------------------------------------------------------------------------------------------
#  Examples:

CALL  GPROD1P_PP.TBL_DELTA	(  12542467796,      --  IN I_RUN_ID DECIMAL(38),
										   NULL,                               --IN I_SCRIPT_ID INTEGER,
											'ECONOMICS_SEARCH',  --IN I_SOURCE_NAME VARCHAR(50),
											'GDEV1T_SEM',                --	IN I_SCHEMA VARCHAR(100),
											'CRA_COMP_REG_UCR',  --IN I_TBL_NAME VARCHAR(100),
											'SELECT * FROM GDEV1V_SEM.TXF_CRA_COMP_REG_UCR', --IN I_SQL_STMT CLOB CHARACTER SET UNICODE,
											'CRA_NUMBER',               --IN I_PRIM_INDEX CLOB CHARACTER SET UNICODE,
											'UCR',                               --	IN I_SECD_INDEX CLOB CHARACTER SET UNICODE,
											NULL,                              --	IN I_NOT_NULLABLE_COL CLOB CHARACTER SET UNICODE,
										    1 ,                                     --	IN I_DELTA   INTEGER ,
											'DM_LAST_UPDATE_DATE', --	IN I_COLS_EXCLUDE_FROM_DELTA VARCHAR(500),
											'CRA_NUMBER',               --	IN I_TARGET_TABLE_KEYS  VARCHAR(500),
											 OUT_RETURN_CODE ,    --OUT O_RETURN_CODE INTEGER,
											 OUT_RETURN_MSG  ,     --OUT O_RETURN_MSG  VARCHAR(10000),
											 OUT_ROWS_COUNT       --	OUT O_ROWS_COUNT INTEGER
									 );
###################################################################################################################################################################
*/									 
BEGIN
	
	DECLARE V_IS_COL_EXIST INTEGER DEFAULT 0;
	DECLARE V_IS_TBL_EXIST INTEGER DEFAULT 0;
	DECLARE V_SCRIPT_ID INTEGER DEFAULT 0;
	DECLARE V_ORG_TBL_NAME VARCHAR(100);
	DECLARE V_TRGT_TABLE_NAME VARCHAR(100);
	DECLARE V_TMP_TABLE_NAME  VARCHAR(100);
	DECLARE V_COLS_EXCL_FROM_DELTA VARCHAR(1000);
	DECLARE V_DELTA_TABLE_NAME_WITH_BUSINESS_COLS VARCHAR(200);
	DECLARE V_DELTA_TABLE_NAME VARCHAR(200);
	DECLARE V_BUSINESS_COLS  VARCHAR(1000);
	DECLARE V_SQL_QUERY	CLOB CHARACTER SET UNICODE;
	DECLARE V_INDEX_QUERY CLOB CHARACTER SET UNICODE;
	DECLARE V_INDEX_QUERY_NO_DATA CLOB CHARACTER SET UNICODE;
	DECLARE V_INDEX_QUERY_FOR_VOLATILE_TBL CLOB CHARACTER SET UNICODE;
	DECLARE V_SECD_INDEX CLOB CHARACTER SET UNICODE;
	DECLARE V_NOT_NULLABLE_COL CLOB CHARACTER SET UNICODE;
	DECLARE V_UPDATE_SET_STMT  VARCHAR(20000);
    DECLARE V_INSERT_COL_STMT  VARCHAR(20000);
	DECLARE V_UPDATE_WHERE_CONDITION   VARCHAR(20000);
	DECLARE V_RETURN_CODE INTEGER DEFAULT 0;
	DECLARE V_RETURN_MSG VARCHAR(10000);
	DECLARE V_ROWS_COUNT INTEGER DEFAULT 0;
	
	DECLARE EXIT HANDLER FOR SQLEXCEPTION 
	BEGIN
		SET V_RETURN_CODE = SQLCODE;
		GET DIAGNOSTICS EXCEPTION 1 V_RETURN_MSG = MESSAGE_TEXT; 
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_RETURN_CODE  = V_RETURN_CODE;
		SET O_ROWS_COUNT = 0;
	END;
	
	SET V_ORG_TBL_NAME = I_TBL_NAME ;
	SET V_TRGT_TABLE_NAME = I_TBL_NAME ;
	SET V_TMP_TABLE_NAME = I_TBL_NAME || '_TMP';
	SET V_DELTA_TABLE_NAME_WITH_BUSINESS_COLS = I_TBL_NAME || '_DELTA_COLS';
	SET V_DELTA_TABLE_NAME = I_TBL_NAME || '_DELTA';
	SET V_SCRIPT_ID = NVL(I_SCRIPT_ID, 1);
	SET V_INDEX_QUERY = I_PRIM_INDEX;
	SET V_SECD_INDEX = I_SECD_INDEX;
	SET V_NOT_NULLABLE_COL = I_NOT_NULLABLE_COL;
	
	IF V_INDEX_QUERY IS NULL
	THEN
		SET V_INDEX_QUERY = 'WITH DATA NO PRIMARY INDEX ;';
		SET V_INDEX_QUERY_NO_DATA = 'WITH NO DATA NO PRIMARY INDEX ;';
		SET V_INDEX_QUERY_FOR_VOLATILE_TBL = 'WITH DATA NO PRIMARY INDEX ON COMMIT PRESERVE ROWS ;';
	ELSE
		SET V_INDEX_QUERY = 'WITH DATA PRIMARY INDEX (' || I_PRIM_INDEX || ')' || CASE WHEN V_SECD_INDEX IS NULL THEN ';' ELSE  V_SECD_INDEX ||' ;'  END ; --' INDEX (' || V_SECD_INDEX || ');' END ;
	    SET V_INDEX_QUERY_NO_DATA = 'WITH NO DATA PRIMARY INDEX (' || I_PRIM_INDEX || ')' || CASE WHEN V_SECD_INDEX IS NULL THEN ';' ELSE V_SECD_INDEX ||';'  END ; 
		SET V_INDEX_QUERY_FOR_VOLATILE_TBL = 'WITH DATA PRIMARY INDEX (' || I_PRIM_INDEX || ')' || CASE WHEN V_SECD_INDEX IS NULL THEN ' ON COMMIT PRESERVE ROWS;' ELSE V_SECD_INDEX ||' ON COMMIT PRESERVE ROWS ;'  END ;
	END IF;
	
	IF I_COLS_EXCLUDE_FROM_DELTA IS NULL
	THEN 
	SET V_COLS_EXCL_FROM_DELTA = 'DM_LAST_UPDATE_DATE';
	ELSE 
	SET V_COLS_EXCL_FROM_DELTA = 'DM_LAST_UPDATE_DATE'|| ','||I_COLS_EXCLUDE_FROM_DELTA ;
	END IF;
	
	SELECT COUNT(*) INTO V_IS_TBL_EXIST  
	FROM DBC.TablesV 
	WHERE TABLENAME = V_TMP_TABLE_NAME
	AND DATABASENAME= I_SCHEMA;
	
	IF V_IS_TBL_EXIST > 0
	THEN
     SET V_SQL_QUERY = ' DROP TABLE '  || I_SCHEMA || '.' || V_TMP_TABLE_NAME  || ';';
	 CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
	END IF ;
	
	SET V_SQL_QUERY = 'CREATE MULTISET TABLE ' || I_SCHEMA || '.' || V_TMP_TABLE_NAME || ' AS (' || I_SQL_STMT || ' ) ' || V_INDEX_QUERY ;
	CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		
	  SELECT COUNT(*) INTO V_IS_COL_EXIST 
	  FROM DBC.TablesV  T INNER JOIN DBC.ColumnsV C ON T.TABLENAME =  C.TABLENAME  AND T.DATABASENAME = C.DATABASENAME
	  WHERE     T.TABLENAME = I_TBL_NAME
	  AND T.DATABASENAME = I_SCHEMA
	  AND C.COLUMNNAME = 'DM_LAST_UPDATE_DATE'
	  AND C.ColumnType = 'TS' 
	  --AND C.ColumnFormat ='YYYY-MM-DDBHH:MI:SS' 
	  ;
	  
	  IF V_IS_COL_EXIST > 0
	THEN
		
	SELECT TRIM ( TRAILING ',' FROM (  XMLAGG(  X.COLUMNNAME ||',' ORDER BY X.COLUMNNAME  )(VARCHAR (10000))))
	  INTO V_BUSINESS_COLS
	  FROM (SELECT DISTINCT COLUMNNAME  
			FROM DBC.ColumnsV
			WHERE TABLENAME = I_TBL_NAME
			AND DATABASENAME = I_SCHEMA
			AND COLUMNNAME NOT IN ( SELECT d.token FROM TABLE (STRTOK_SPLIT_TO_TABLE(1, V_COLS_EXCL_FROM_DELTA , ',')
																	RETURNS (outkey INTEGER, tokennum INTEGER, token VARCHAR(100) CHARACTER SET UNICODE) ) AS d ) 
			) X ;
	 
	 --DROP DELTA_TABLE_NAME_WITH_BUSINESS_COLS TABLE IF IT IS ALREADRY EXIST
	SELECT COUNT(*) INTO V_IS_TBL_EXIST  FROM DBC.TablesV WHERE TABLENAME = V_DELTA_TABLE_NAME_WITH_BUSINESS_COLS;
	IF V_IS_TBL_EXIST > 0
	THEN
		SET V_SQL_QUERY = 'DROP TABLE ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME_WITH_BUSINESS_COLS;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		SET O_RETURN_CODE = V_RETURN_CODE;
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_ROWS_COUNT = V_ROWS_COUNT;
		SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
	
	END IF;
	--CREATE DELTA_TABLE_NAME_WITH_BUSINESS_COLS TABLE
	
		SET V_SQL_QUERY = 'CREATE MULTISET TABLE ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME_WITH_BUSINESS_COLS || ' AS ( SEL '||V_BUSINESS_COLS||' FROM ' || I_SCHEMA || '.'|| V_TMP_TABLE_NAME || ' MINUS SEL '||V_BUSINESS_COLS||' FROM '||I_SCHEMA||'.'||I_TBL_NAME||' ) ' ||V_INDEX_QUERY;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		SET O_RETURN_CODE = V_RETURN_CODE;
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_ROWS_COUNT = V_ROWS_COUNT;
		SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
		

	
		 --DROP DELTA_TABLE_NAME TABLE IF IT IS ALREADRY EXIST
	SELECT COUNT(*) INTO V_IS_TBL_EXIST  FROM DBC.TablesV WHERE TABLENAME = V_DELTA_TABLE_NAME;
	IF V_IS_TBL_EXIST > 0
	THEN
		SET V_SQL_QUERY = 'DROP TABLE ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		SET O_RETURN_CODE = V_RETURN_CODE;
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_ROWS_COUNT = V_ROWS_COUNT;
		SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
		
	END IF;
	
		--CREATE DELTA_TABLE_NAME TABLE
	
		SET V_SQL_QUERY = 'CREATE MULTISET TABLE ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME || ' AS ( SEL * FROM ' || I_SCHEMA || '.'|| V_TMP_TABLE_NAME ||')' ||V_INDEX_QUERY_NO_DATA; 
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		SET O_RETURN_CODE = V_RETURN_CODE;
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_ROWS_COUNT = V_ROWS_COUNT;
		SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
		
	 --INSERT  DATA INTO  DELTA_TABLE_NAME TABLE
	 SET V_SQL_QUERY = 'INSERT INTO  ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME || '  SEL * FROM ' || I_SCHEMA || '.'|| V_TMP_TABLE_NAME || 
		'  WHERE ('||I_TARGET_TABLE_KEYS||') IN (SEL '||I_TARGET_TABLE_KEYS||' FROM ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME_WITH_BUSINESS_COLS ||')
		QUALIFY ROW_NUMBER() OVER(PARTITION BY '||I_TARGET_TABLE_KEYS||' ORDER BY '||I_TARGET_TABLE_KEYS||' DESC)=1 ; ' ;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		SET O_RETURN_CODE = V_RETURN_CODE;
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_ROWS_COUNT = V_ROWS_COUNT;
		SET V_SCRIPT_ID = V_SCRIPT_ID + 1;


    IF   V_ROWS_COUNT > 0 
	THEN
	-- DELETE KEYS NOT EXIST IN VOLATILE TABLE
	IF V_RETURN_CODE = 0
	THEN
		SET V_SQL_QUERY = 'DELETE  ' ||I_SCHEMA||'.'||I_TBL_NAME || ' WHERE ( '||I_TARGET_TABLE_KEYS|| ' ) NOT IN ( SEL '||I_TARGET_TABLE_KEYS||' FROM ' || I_SCHEMA || '.'|| V_TMP_TABLE_NAME ||' );';
				CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
				SET O_RETURN_CODE = V_RETURN_CODE;
				SET O_RETURN_MSG = V_RETURN_MSG;
				SET O_ROWS_COUNT = V_ROWS_COUNT;
				SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
		END IF;

	--UPDATE DATA IF EXIST USING KEY TABLE
	  IF V_RETURN_CODE = 0
		THEN
		SEL TRIM ( TRAILING ',' FROM (  XMLAGG(  X.U ||',' ORDER BY X.U  )(VARCHAR (20000)))) A
		INTO V_UPDATE_SET_STMT
			FROM (
			SELECT DISTINCT  CASE WHEN COLUMNNAME  = 'DM_LAST_UPDATE_DATE' THEN  COLUMNNAME ||'= CURRENT_TIMESTAMP(0)'
									ELSE COLUMNNAME ||' = '||  I_SCHEMA || '.' || V_DELTA_TABLE_NAME ||'.'|| COLUMNNAME 
									END AS U
				FROM DBC.ColumnsV
				WHERE TABLENAME = I_TBL_NAME
				AND DATABASENAME = I_SCHEMA
				AND COLUMNNAME NOT IN ( SELECT d.token FROM TABLE (STRTOK_SPLIT_TO_TABLE(1, I_TARGET_TABLE_KEYS , ',')
																		RETURNS (outkey INTEGER, tokennum INTEGER, token VARCHAR(100) CHARACTER SET UNICODE) ) AS d ) 
			)X;

		SEL SUBSTR (Z.COND,1,LENGTH(Z.COND)-4)
		INTO V_UPDATE_WHERE_CONDITION
		FROM (SELECT TRIM ( TRAILING ',' FROM (  XMLAGG(  X."CONDITION" ||' AND ' ORDER BY X."CONDITION"  )(VARCHAR (10000)))) AS COND
		  FROM (SELECT DISTINCT I_SCHEMA||'.'||I_TBL_NAME||'.'||COLUMNNAME ||' = ' ||  I_SCHEMA || '.' || V_DELTA_TABLE_NAME ||'.'||COLUMNNAME  AS "CONDITION"
				FROM DBC.ColumnsV
				WHERE TABLENAME = I_TBL_NAME
				AND DATABASENAME = I_SCHEMA
				AND COLUMNNAME  IN ( SELECT d.token FROM TABLE (STRTOK_SPLIT_TO_TABLE(1, I_TARGET_TABLE_KEYS , ',')
																		RETURNS (outkey INTEGER, tokennum INTEGER, token VARCHAR(100) CHARACTER SET UNICODE) ) AS d ) 
				) X
				) Z;
			
			SET V_SQL_QUERY = 'UPDATE  ' ||I_SCHEMA||'.'||I_TBL_NAME || ' SET '||V_UPDATE_SET_STMT || ' WHERE ( '||V_UPDATE_WHERE_CONDITION|| ' ) ;';
			CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
			SET O_RETURN_CODE = V_RETURN_CODE;
			SET O_RETURN_MSG = V_RETURN_MSG;
			SET O_ROWS_COUNT = V_ROWS_COUNT;
			SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
		END IF; 
		
		--INSERT DATA IF NOT EXIST USING KEY TABLE
		   IF V_RETURN_CODE = 0
		THEN
		
		   SEL TRIM ( TRAILING ',' FROM (  XMLAGG(  X.COLUMNNAME ||',' ORDER BY X.COLUMNNAME)(VARCHAR (20000)))) A
		INTO V_INSERT_COL_STMT
			FROM (
			SELECT DISTINCT   COLUMNNAME  
				FROM DBC.ColumnsV
				WHERE TABLENAME = I_TBL_NAME
				AND DATABASENAME = I_SCHEMA
				AND COLUMNNAME <> 'DM_LAST_UPDATE_DATE'
			)X;
			
			SET V_SQL_QUERY = 'INSERT INTO   ' ||I_SCHEMA||'.'||I_TBL_NAME || '( '||V_INSERT_COL_STMT||' )'  || ' SEL '|| V_INSERT_COL_STMT ||' FROM '|| I_SCHEMA || '.' || V_DELTA_TABLE_NAME || ' WHERE ( '||I_TARGET_TABLE_KEYS|| ' ) NOT IN ( SEL '||I_TARGET_TABLE_KEYS||' FROM ' || I_SCHEMA || '.' || I_TBL_NAME ||' );';
			CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
			SET O_RETURN_CODE = V_RETURN_CODE;
			SET O_RETURN_MSG = V_RETURN_MSG;
			SET O_ROWS_COUNT = V_ROWS_COUNT;
			SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
		END IF; 
	END IF;
	/*IF V_RETURN_CODE = 0
	THEN
	SET V_SQL_QUERY = 'DROP TABLE ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME_WITH_BUSINESS_COLS;
		CALL GDEV1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		SET O_RETURN_CODE = V_RETURN_CODE;
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_ROWS_COUNT = V_ROWS_COUNT;
		SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
		
		SET V_SQL_QUERY = 'DROP TABLE ' || I_SCHEMA || '.' || V_DELTA_TABLE_NAME;
		CALL GDEV1P_PP.DBC_SYSEXECSQL_WITH_LOG(V_SQL_QUERY, V_SCRIPT_ID, I_RUN_ID, COALESCE(I_SOURCE_NAME, I_TBL_NAME), I_TBL_NAME, NULL ,1 , V_RETURN_CODE, V_RETURN_MSG, V_ROWS_COUNT);
		SET O_RETURN_CODE = V_RETURN_CODE;
		SET O_RETURN_MSG = V_RETURN_MSG;
		SET O_ROWS_COUNT = V_ROWS_COUNT;
		SET V_SCRIPT_ID = V_SCRIPT_ID + 1;
	
	 END IF;*/
	 
	ELSE 
		SET O_RETURN_CODE =  -1;
		SET O_RETURN_MSG =  'DM_LAST_UPDATE_DATE NOT EXISTS IN TABLE';
	END IF;	
END;

REPLACE PROCEDURE GPROD1P_PP.TBL_CLONE	(    IN IN_RUN_ID DECIMAL(38),
																																IN IN_SCRIPT_ID INTEGER,
																															 	IN IN_SOURCE_NAME VARCHAR(50),
																															 	IN IN_SCHEMA VARCHAR(100),
																															 	IN IN_TBL_NAME VARCHAR(100),
																																IN IN_SQL_STMT CLOB CHARACTER 
SET	UNICODE,
																																IN IN_PRIM_INDEX CLOB CHARACTER 
SET	UNICODE,
																																IN IN_SECD_INDEX CLOB CHARACTER 
SET	UNICODE,
																																IN IN_NOT_NULLABLE_COL CLOB CHARACTER 
SET	UNICODE,
																															 	OUT OUT_RETURN_CODE INTEGER,
																															 	OUT OUT_RETURN_MSG  VARCHAR(10000),
																															 	OUT OUT_ROWS_COUNT INTEGER
																														 )
BEGIN
	
	DECLARE IS_TBL_EXIST INTEGER DEFAULT 0;
	DECLARE SCRIPT_ID INTEGER DEFAULT 0;
	DECLARE RETURN_CODE INTEGER DEFAULT 0;
	DECLARE RETURN_MSG VARCHAR(10000);
	DECLARE ROWS_COUNT INTEGER DEFAULT 0;
	DECLARE NEW_ORG_TBL_NAME VARCHAR(100);
	DECLARE TMP_TABLE_NAME VARCHAR(100);
	DECLARE SQL_QUERY	CLOB CHARACTER 
SET	UNICODE;
	DECLARE INDEX_QUERY CLOB CHARACTER 
SET	UNICODE;
	DECLARE SECD_INDEX CLOB CHARACTER 
SET	UNICODE;
	DECLARE NOT_NULLABLE_COL CLOB CHARACTER 
SET	UNICODE;
	
	DECLARE EXIT HANDLER FOR SQLEXCEPTION 
	BEGIN
		SET RETURN_CODE = SQLCODE;
		GET DIAGNOSTICS EXCEPTION 1 RETURN_MSG = MESSAGE_TEXT; 
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_RETURN_CODE  = RETURN_CODE;
		SET OUT_ROWS_COUNT = 0;
	END;
	
	SET NEW_ORG_TBL_NAME = IN_TBL_NAME || '_DEL';
	SET TMP_TABLE_NAME = IN_TBL_NAME || '_TMP';
	SET SCRIPT_ID = NVL(IN_SCRIPT_ID, 1);
	SET INDEX_QUERY = IN_PRIM_INDEX;
	SET SECD_INDEX = IN_SECD_INDEX;
	SET NOT_NULLABLE_COL = IN_NOT_NULLABLE_COL;
	
	IF INDEX_QUERY IS NULL
	THEN
		SET INDEX_QUERY = 'WITH DATA NO PRIMARY INDEX;';
	ELSE
		SET INDEX_QUERY = 'WITH DATA PRIMARY INDEX (' || IN_PRIM_INDEX || ')' || 
CASE	
	WHEN SECD_INDEX IS NULL THEN ';' 
	ELSE SECD_INDEX || ';' 
END	; --' INDEX (' || SECD_INDEX || ');' END ;
	END IF;
	
	--DROP _TMP TABLE IF IT IS ALREADRY EXIST
	SELECT COUNT(*) INTO IS_TBL_EXIST  
FROM	DBC.TablesV 
WHERE	TABLENAME = TMP_TABLE_NAME 
	and databasename= IN_SCHEMA;
	IF IS_TBL_EXIST > 0
	THEN
		SET SQL_QUERY = 'DROP TABLE ' || IN_SCHEMA || '.' || TMP_TABLE_NAME;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(SQL_QUERY, SCRIPT_ID,
		IN_RUN_ID, COALESCE(IN_SOURCE_NAME, IN_TBL_NAME), IN_TBL_NAME,
		NULL ,1 , RETURN_CODE, RETURN_MSG, ROWS_COUNT);
		SET OUT_RETURN_CODE = RETURN_CODE;
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_ROWS_COUNT = ROWS_COUNT;
		SET SCRIPT_ID = SCRIPT_ID + 1;
	END IF;
	
	--DROP _DEL TABLE IF IT IS ALREADRY EXIST
	SELECT COUNT(*) INTO IS_TBL_EXIST  
FROM	DBC.TablesV 
WHERE	TABLENAME = NEW_ORG_TBL_NAME 
	and databasename= IN_SCHEMA;
	IF IS_TBL_EXIST > 0 
	AND RETURN_CODE = 0
	THEN
		SET SQL_QUERY = 'DROP TABLE ' || IN_SCHEMA || '.' || NEW_ORG_TBL_NAME;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(SQL_QUERY, SCRIPT_ID,
		IN_RUN_ID, COALESCE(IN_SOURCE_NAME, IN_TBL_NAME), IN_TBL_NAME,
		NULL ,1 , RETURN_CODE, RETURN_MSG, ROWS_COUNT);
		SET OUT_RETURN_CODE = RETURN_CODE;
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_ROWS_COUNT = ROWS_COUNT;
		SET SCRIPT_ID = SCRIPT_ID + 1;
	END IF;
	
	--CREATE _TMP TABLE
	IF RETURN_CODE = 0
	THEN
		SET SQL_QUERY = 'CREATE MULTISET TABLE ' || IN_SCHEMA || '.' || TMP_TABLE_NAME || ' AS (' || IN_SQL_STMT || ' ) ' || INDEX_QUERY;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(SQL_QUERY, SCRIPT_ID,
		IN_RUN_ID, COALESCE(IN_SOURCE_NAME, IN_TBL_NAME), IN_TBL_NAME,
		NULL ,1 , RETURN_CODE, RETURN_MSG, ROWS_COUNT);
		SET OUT_RETURN_CODE = RETURN_CODE;
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_ROWS_COUNT = ROWS_COUNT;
		SET SCRIPT_ID = SCRIPT_ID + 1;
	END IF;
	
   WHILE (LENGTH(NOT_NULLABLE_COL) > 0 )
	DO
		SET SQL_QUERY = 'ALTER TABLE ' || IN_SCHEMA || '.' || TMP_TABLE_NAME || ' ADD ' || STRTOK(NOT_NULLABLE_COL,
		';', 1) || ' NOT NULL;' ;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(SQL_QUERY, SCRIPT_ID,
		IN_RUN_ID, COALESCE(IN_SOURCE_NAME, IN_TBL_NAME), IN_TBL_NAME,
		NULL ,1 , RETURN_CODE, RETURN_MSG, ROWS_COUNT);
		SET OUT_RETURN_CODE = RETURN_CODE;
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_ROWS_COUNT = ROWS_COUNT;
		SET SCRIPT_ID = SCRIPT_ID + 1;
		
		SET NOT_NULLABLE_COL = OREPLACE(NOT_NULLABLE_COL, STRTOK(NOT_NULLABLE_COL,
		';', 1) || ';', '');
	END WHILE;
	
	--RENAME ORG TABLE TO TBLNAME_DEL
	SELECT COUNT(*) INTO IS_TBL_EXIST  
FROM	DBC.TablesV 
WHERE	TABLENAME = IN_TBL_NAME 
	and databasename= IN_SCHEMA;
	IF IS_TBL_EXIST > 0 
	AND RETURN_CODE = 0
	THEN
		SET SQL_QUERY = 'RENAME TABLE ' || IN_SCHEMA || '.' || IN_TBL_NAME || ' TO ' || IN_SCHEMA || '.' || NEW_ORG_TBL_NAME ;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(SQL_QUERY, SCRIPT_ID,
		IN_RUN_ID, COALESCE(IN_SOURCE_NAME, IN_TBL_NAME), IN_TBL_NAME,
		NULL ,1 , RETURN_CODE, RETURN_MSG, ROWS_COUNT);
		SET OUT_RETURN_CODE = RETURN_CODE;
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_ROWS_COUNT = ROWS_COUNT;
		SET SCRIPT_ID = SCRIPT_ID + 1;
	END IF;
	
	--RENAME _TMP TO ORG TABLE NAME
	IF RETURN_CODE = 0
	THEN
		SET SQL_QUERY = 'RENAME TABLE ' || IN_SCHEMA || '.' || TMP_TABLE_NAME || ' TO ' || IN_SCHEMA || '.' || IN_TBL_NAME ;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(SQL_QUERY, SCRIPT_ID,
		IN_RUN_ID, COALESCE(IN_SOURCE_NAME, IN_TBL_NAME), IN_TBL_NAME,
		NULL ,1 , RETURN_CODE, RETURN_MSG, ROWS_COUNT);
		SET OUT_RETURN_CODE = RETURN_CODE;
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_ROWS_COUNT = ROWS_COUNT;
		SET SCRIPT_ID = SCRIPT_ID + 1;
	END IF;
	
	--DELETE _DEL TABLE
	SELECT COUNT(*) INTO IS_TBL_EXIST  
FROM	DBC.TablesV 
WHERE	TABLENAME = NEW_ORG_TBL_NAME 
	and databasename= IN_SCHEMA;
	IF IS_TBL_EXIST > 0 
	AND RETURN_CODE = 0
	THEN
		SET SQL_QUERY = 'DROP TABLE ' || IN_SCHEMA || '.' || NEW_ORG_TBL_NAME;
		CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(SQL_QUERY, SCRIPT_ID,
		IN_RUN_ID, COALESCE(IN_SOURCE_NAME, IN_TBL_NAME), IN_TBL_NAME,
		NULL ,1 , RETURN_CODE, RETURN_MSG, ROWS_COUNT);
		SET OUT_RETURN_CODE = RETURN_CODE;
		SET OUT_RETURN_MSG = RETURN_MSG;
		SET OUT_ROWS_COUNT = ROWS_COUNT;
	END IF;
			
END;

REPLACE PROCEDURE GPROD1P_PP.SP_POPULATE_TBL	 (	  IN IN_PRIORITY INTEGER,
																																				  IN IN_RUN_ID DECIMAL(38),
																																				  IN IN_SOURCE_NAME VARCHAR(100),
																																				  OUT OUT_RETURN_CODE INTEGER,
																													 						 	  OUT OUT_RETURN_MSG  VARCHAR(10000),
																													 						 	  OUT OUT_ROWS_COUNT INTEGER
																																			 )																																							
BEGIN
	  DECLARE RUN_ID DECIMAL(38);
	  DECLARE RETURN_CODE INTEGER DEFAULT 0;
	  DECLARE RETURN_MSG VARCHAR(10000);
	  DECLARE ROWS_COUNT INTEGER DEFAULT 0;
	  
	  DECLARE EXIT HANDLER FOR SQLEXCEPTION 
	  BEGIN
			SET RETURN_CODE = SQLCODE;
			GET DIAGNOSTICS EXCEPTION 1 RETURN_MSG = MESSAGE_TEXT; 
			SET OUT_RETURN_MSG = RETURN_MSG;
			SET OUT_RETURN_CODE  = RETURN_CODE;
			SET OUT_ROWS_COUNT = 0;
	  END;
	  
	  SET RUN_ID = IN_RUN_ID;
	  IF RUN_ID IS NULL 
	  THEN 
			SELECT GPROD1P_FF.GENERATE_RUN_ID() INTO RUN_ID;
	  END IF;
	  
	  FOR P_CUR AS COLUMN_CURSOR CURSOR FOR
	  SELECT	 DISTINCT PLS_SRC_NAME
	  FROM		GPROD1T_GCFR.AUTOMATION_SCRIPTS
	  WHERE	( PLS_SRC_NAME = IN_SOURCE_NAME OR  IN_SOURCE_NAME IS NULL ) AND
	  						 ( PLS_PRIORITY = IN_PRIORITY OR IN_PRIORITY IS NULL )
							 AND PLS_IS_ACTIVE = 1
		ORDER BY
							 PLS_PRIORITY
		DO
				  
				  SET RETURN_CODE = 0;
				  SPLABEL_1:
				  FOR CUR_1 AS COLUMN_CURSOR CURSOR FOR
			     SELECT	 * 
				  FROM		GPROD1T_GCFR.AUTOMATION_SCRIPTS
				  WHERE	PLS_SRC_NAME = P_CUR.PLS_SRC_NAME
				  AND PLS_IS_ACTIVE = 1
				  ORDER BY
										 PLS_SRC_SEQ
			     DO 
				  		IF RETURN_CODE = 0
						THEN
							IF CUR_1.IS_IMMEDIATE_QRY = 0
								  THEN
								   IF  CUR_1.DELTA = 1 AND CUR_1.TARGET_TABLE_KEYS IS NOT NULL
									THEN
									CALL GPROD1P_PP.TBL_DELTA	(   RUN_ID,
																	NULL,
																	CUR_1.PLS_SRC_NAME,
																	CUR_1.PLS_SCHEMA,
																	CUR_1.PLS_TBL_NAME,
																	CUR_1.PLS_SQL_STMT,
																	CUR_1.PLS_PRIM_INDEX,
																	CUR_1.PLS_SECD_INDEX,
																	CUR_1.NOT_NULLABLE_COL,
																	CUR_1.DELTA ,
																	CUR_1.COLS_EXCLUDE_FROM_DELTA ,
																	CUR_1.TARGET_TABLE_KEYS,
																	RETURN_CODE,
																	RETURN_MSG,
																	ROWS_COUNT 
															 );
								  ELSE	
							  		CALL  GPROD1P_PP.TBL_CLONE	( 				RUN_ID,
																				NULL,
																				CUR_1.PLS_SRC_NAME,
																				CUR_1.PLS_SCHEMA,
																				CUR_1.PLS_TBL_NAME,
																				CUR_1.PLS_SQL_STMT,
																				CUR_1.PLS_PRIM_INDEX,
																				CUR_1.PLS_SECD_INDEX,
																				CUR_1.NOT_NULLABLE_COL,
																				RETURN_CODE,
																				RETURN_MSG,
																				ROWS_COUNT 
																			);
									END IF;										
							ELSE
									CALL GPROD1P_PP.DBC_SYSEXECSQL_WITH_LOG(	CUR_1.PLS_SQL_STMT,
																																							CUR_1.PLS_SRC_SEQ,
																																							RUN_ID,
																																							CUR_1.PLS_SRC_NAME,
																																							CUR_1.PLS_TBL_NAME,
																																							NULL ,
																																							1 ,
																																							RETURN_CODE,
																																							RETURN_MSG,
																																							ROWS_COUNT);
							END IF;
							
							IF CUR_1.IGNORING_ERR_FLAG = 0
							THEN
								SET OUT_RETURN_CODE = RETURN_CODE;
							ELSE
								SET RETURN_CODE = 0;
							END IF;
							
							SET OUT_RETURN_MSG = RETURN_MSG;
							SET OUT_ROWS_COUNT = ROWS_COUNT;
						ELSE
							LEAVE SPLABEL_1;
					END IF;
			    END FOR;
		END FOR;
		
END;
