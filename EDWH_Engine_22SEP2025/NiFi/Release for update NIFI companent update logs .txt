UPDATE TGT 
FROM ${CONFIG_SCHEMA}.EXEC_SOURCE_LOGS TGT, ${CONFIG_SCHEMA}.V_EXEC_SOURCE_LOAD_LOGS SRC
SET 
	 STG_DONE =  LAYER_DONE
	,ERROR_CODE = CASE WHEN COALESCE(ERROR_CODE,0) = 0 THEN LAST_ERROR_CODE ELSE ERROR_CODE END
	,ERROR_MSG = CASE WHEN COALESCE(ERROR_CODE,0) = 0 THEN LAST_ERROR_MSG ELSE ERROR_MSG END 
WHERE SRC.RUN_ID = TGT.RUN_ID
AND SRC.LOAD_ID = TGT.LOAD_ID
AND SRC.SOURCE_NAME = TGT.SOURCE_NAME
AND SRC.LAYER_NAME = 'STG'
--AND (TGT.END_TIMESTAMP IS NULL OR TGT.DONE = 0)
AND STG_DONE <>  LAYER_DONE;

UPDATE TGT 
FROM ${CONFIG_SCHEMA}.EXEC_SOURCE_LOGS TGT, 
(
			select RUN_ID,  LOAD_ID, SOURCE_NAME
			
			, 	SUM(ROWS_INSERTED_COUNT) 		ROWS_INSERTED_COUNT
			,	sum(ROWS_UPDATED_COUNT) 		ROWS_UPDATED_COUNT
			,	sum(ROWS_DELETED_COUNT) 		ROWS_DELETED_COUNT
			from ${CONFIG_SCHEMA}.V_EXEC_SOURCE_LOAD_LOGS 
			group by RUN_ID,  LOAD_ID, SOURCE_NAME
			having SUM(case when PRECEDING_LAYER_DONE = 1 and PRECEDING_FAILED_CNT > 0 then 0 else CNT_PROCESSES end) = sum(passed) + sum(failed)
) SRC
SET 
	ROWS_INSERTED_COUNT = src.ROWS_INSERTED_COUNT
	,ROWS_UPDATED_COUNT = src.ROWS_UPDATED_COUNT
	,ROWS_DELETED_COUNT = src.ROWS_DELETED_COUNT
	,DONE =  1
	,END_TIMESTAMP = current_timestamp
WHERE SRC.RUN_ID = TGT.RUN_ID
AND SRC.LOAD_ID = TGT.LOAD_ID
AND SRC.SOURCE_NAME = TGT.SOURCE_NAME
AND (TGT.END_TIMESTAMP IS NULL OR TGT.DONE = 0);

update tgt
from ${CONFIG_SCHEMA}.CDC_AUDIT tgt, ${CONFIG_SCHEMA}.EXEC_SOURCE_LOGS src
set PROCESSED = 1, UPDATED_BY = user, UPDATED_AT=current_timestamp
where SRC.LOAD_ID = TGT.LOAD_ID
AND SRC.SOURCE_NAME = TGT.SOURCE_NAME
and src.DONE = 1
and src.ERROR_CODE = 0
and tgt.PROCESSED = 0;
